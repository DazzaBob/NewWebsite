    // -------------------------------
    // ADDRESS MODAL
    // -------------------------------
    Website.AddressModal = (function () {
        let modal, dialog, btnClose, btnCancel, btnSave, btnToggle, form, untrap, lastFocused;
        let isOpen = false;

        // -----------------------
        // UTILITY FUNCTIONS
        // -----------------------
        const $ = (sel, root = document) => root.querySelector(sel);
        const $$ = (sel, root = document) => Array.from(root.querySelectorAll(sel));
        const show = (el, v = true) => { if (el) el.style.display = v ? 'block' : 'none'; };
        const disable = (el, v = true) => { if (el) el.disabled = v; };
        const trapFocus = (container) => {
            const FOCUS_SEL = 'button,[href],input,select,textarea,[tabindex]:not([tabindex="-1"])';
            const focusables = () => $$(FOCUS_SEL, container).filter(el => !el.hasAttribute('disabled') && el.offsetParent !== null);
            const handler = e => {
                if (e.key !== 'Tab') return;
                const f = focusables();
                if (!f.length) { e.preventDefault(); return; }
                const first = f[0], last = f[f.length - 1];
                if (e.shiftKey && document.activeElement === first) { e.preventDefault(); last.focus(); }
                else if (!e.shiftKey && document.activeElement === last) { e.preventDefault(); first.focus(); }
            };
            document.addEventListener('keydown', handler, true);
            return () => document.removeEventListener('keydown', handler, true);
        };
        const onEsc = e => { if (e.key === 'Escape') { e.preventDefault(); close(); } };

        // -----------------------
        // MODAL ACTIONS
        // -----------------------
        async function loadAddresses() {
            const container = $('#addressListContainer') || $('.custom-modal-body', dialog);
            if (!container) return;

            container.innerHTML = '<div class="loading-spinner" aria-live="polite">Loading…</div>';

            try {
                const res = await fetch('/User/Address/ListPartial', { method: 'GET', credentials: 'same-origin' });
                if (!res.ok) throw new Error('Failed to load addresses.');
                container.innerHTML = await res.text();
                form = $('#addressForm') || form; // refresh reference if replaced
            } catch (err) {
                container.innerHTML = `<p class="error" role="alert">${err.message || 'Unable to load addresses.'}</p>`;
            }
        }

        function open() {
            if (isOpen) return;

            Website.UI.forceStopAllLoading();

            isOpen = true;
            lastFocused = document.activeElement;

            modal = modal || $('#addressModal');
            if (!modal) return;
            dialog = dialog || modal.querySelector('.custom-modal-content');
            btnClose = btnClose || $('#addressClose');
            btnCancel = btnCancel || $('#addressCancel');
            btnSave = btnSave || $('#addressSave');
            btnToggle = btnToggle || $('#addressToggle');
            form = form || $('#addressForm');

            show(modal, true);
            document.body.style.overflow = 'hidden';
            untrap = trapFocus(dialog);
            (dialog.querySelector('input[type="radio"]:checked,button,[tabindex]') || dialog).focus();
            document.addEventListener('keydown', onEsc, true);

            loadAddresses();
        }

        function close() {
            if (!isOpen) return;
            Website.UI.forceStopAllLoading();

            isOpen = false;
            show(modal, false);
            document.body.style.overflow = '';
            untrap && untrap();
            document.removeEventListener('keydown', onEsc, true);
            (lastFocused || btnToggle)?.focus();
        }

        async function onSubmit(e) {
            e.preventDefault();
            const sel = form.querySelector('input[name="selectedAddress"]:checked');
            if (!sel) return;

            disable(btnSave, true);
            const oldText = btnSave.textContent;
            btnSave.textContent = 'Saving…';

            try {
                const res = await fetch('/User/Address/SetDefault', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ id: sel.value })
                });
                if (!res.ok) throw new Error('Save failed');
                const data = await res.json();
                const lbl = (data?.label || '').trim();
                if (lbl && btnToggle) {
                    btnToggle.innerHTML = `<i class="fa-solid fa-home"></i> ${lbl}&nbsp;<i class="fa-solid fa-chevron-down dropdown-icon"></i>`;
                }
                close();
            } catch (err) {
                console.error('Save failed', err);
            } finally {
                btnSave.textContent = oldText;
                disable(btnSave, false);
                Website.UI.forceStopAllLoading();
            }
        }

        function init() {
            modal = $('#addressModal');
            if (!modal) return;

            btnToggle = $('#addressToggle');
            on(btnToggle, 'click', open);
            btnClose = $('#addressClose');
            btnCancel = $('#addressCancel');
            on(btnClose, 'click', close);
            on(btnCancel, 'click', close);
            on(modal, 'click', e => { if (e.target === modal) close(); });
            form = $('#addressForm');
            form && on(form, 'submit', onSubmit);

            // Preload default label from query string
            const qs = new URLSearchParams(location.search);
            const lbl = qs.get('defaultLabel');
            if (lbl && btnToggle) {
                btnToggle.innerHTML = `<i class="fa-solid fa-home"></i> ${lbl}&nbsp;<i class="fa-solid fa-chevron-down dropdown-icon"></i>`;
                history.replaceState(null, '', location.pathname + location.hash);
            }
        }

        return { init };
    })();

    // -------------------------------
    // ADDRESS UI
    // -------------------------------
    (function AddressUI() {
        const addLink = document.getElementById("add-address-link");
        const newCard = document.getElementById("new-address-card");
        const cancelBtn = document.getElementById("cancelNewAddress");

        if (!addLink || !newCard) return; // nothing to do if elements not on this page

        addLink.addEventListener("click", (e) => {
            e.preventDefault();
            addLink.style.display = "none";
            newCard.style.display = "block";
        });

        if (cancelBtn) {
            cancelBtn.addEventListener("click", () => {
                newCard.style.display = "none";
                addLink.style.display = "inline-block";
            });
        }
    })();
